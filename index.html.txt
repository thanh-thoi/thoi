    <section class="grid md:grid-cols-3 gap-6">
      <div class="bg-white p-5 rounded-2xl border border-[#800020]/15 shadow-sm space-y-3">
        <h2 class="font-bold text-[#800020]">Hồ sơ sinh viên</h2>
        <input id="student-name" type="text" placeholder="Họ và tên" class="w-full px-3 py-2 rounded-lg bg-[#FCF9F7] border border-[#800020]/20 text-sm" />
        <input id="student-id" type="text" placeholder="Mã số sinh viên" class="w-full px-3 py-2 rounded-lg bg-[#FCF9F7] border border-[#800020]/20 text-sm font-mono" />
        <button onclick="saveProfile()" class="w-full py-2 rounded-lg bg-[#800020] text-white text-sm font-semibold">Lưu hồ sơ</button>
      </div>

      <div class="md:col-span-2 bg-white p-5 rounded-2xl border border-[#800020]/15 shadow-sm space-y-3">
        <h2 class="font-bold text-[#800020]">Danh sách chuyên đề</h2>
        <div id="topic-list" class="space-y-2 text-sm"></div>
      </div>
    </section>

  <script>
    const topics = [
      "1. Tổng quan CMCN 4.0 & Chuyển đổi số",
      "2. Kiến trúc phần cứng máy tính",
      "3. Hệ điều hành & Ảo hóa Docker",
      "4. Điện toán đám mây",
      "5. Mạng di động 6G & IoT",
      "6. Tự động hóa quy trình",
      "7. Prompt Engineering & An toàn số"
    ];
    const done = JSON.parse(localStorage.getItem("doneTopics") || "[]");

    function renderTopics() {
      const box = document.getElementById("topic-list");
      box.innerHTML = topics.map((t, idx) => {
        const checked = done.includes(idx) ? "checked" : "";
        return `<label class="flex items-center gap-2 p-2 rounded-lg hover:bg-[#FCF9F7] cursor-pointer">
          <input type="checkbox" ${checked} onchange="toggleTopic(${idx})" class="accent-[#800020]" />
          <span>${t}</span>
        </label>`;
      }).join("");
      document.getElementById("progress-label").textContent = `${done.length} / ${topics.length} Chuyên đề`;
    }

    function toggleTopic(idx) {
      const pos = done.indexOf(idx);
      if (pos >= 0) done.splice(pos, 1);
      else done.push(idx);
      localStorage.setItem("doneTopics", JSON.stringify(done));
      renderTopics();
    }

    function saveProfile() {
      localStorage.setItem("studentProfile", JSON.stringify({
        name: document.getElementById("student-name").value,
        id: document.getElementById("student-id").value
      }));
      alert("Đã lưu hồ sơ trên trình duyệt.");
    }

    const saved = JSON.parse(localStorage.getItem("studentProfile") || "{}");
    if (saved.name) document.getElementById("student-name").value = saved.name;
    if (saved.id) document.getElementById("student-id").value = saved.id;
    renderTopics();
  </script>
